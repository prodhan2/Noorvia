### অনলাইন আযান সিস্টেম
```dart
// OnlineAzanList ক্লাস
class OnlineAzanList {
  static const String baseUrl = 'https://www.islamcan.com/audio/adhan/';
  
  static List<Map<String, String>> getAzanList() {
    return List.generate(20, (index) {
      final num = index + 1;
      return {
        'url': '${baseUrl}azan$num.mp3',
        'name': 'আযান $num',
        'id': 'azan$num',
      };
    });
  }
}
```
